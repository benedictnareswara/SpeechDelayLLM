# Build-time tooling — install on your laptop only, never on the device.
#
#   pip install -r tools/requirements-tools.txt
#
# These render the phrase bank and draft new phrases. The Orange Pi has no
# network at runtime and needs none of them.

# ── Text-to-speech for rendering the bank ────────────────────
# Free, no key, unofficial endpoint. Good enough for a prototype.
gTTS>=2.5.3

# Better id-ID voices (Wavenet/Neural2). Needs a service account and billing:
#   python tools/render_bank.py --voice google-cloud
google-cloud-texttospeech>=2.16.0

# ── Phrase authoring ─────────────────────────────────────────
# tools/expand_bank.py drafts new template variants for human review.
google-generativeai>=0.8.0
python-dotenv>=1.0.1
